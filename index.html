<script>
/*
=========================================
CONFIGURATION
=========================================
*/
const VERSION = "15.18.1";
const DATA_URL =
`https://ddragon.leagueoflegends.com/cdn/${VERSION}/data/fr_FR/champion.json`;
const IMAGE_URL =
`https://ddragon.leagueoflegends.com/cdn/${VERSION}/img/champion/`;
const SPLASH_URL =
`https://ddragon.leagueoflegends.com/cdn/img/champion/splash/`;
let allChampions = [];
let currentRole = "Tous";
/*
=========================================
CHARGER LES CHAMPIONS
=========================================
*/
async function loadChampions() {
try {
const response = await fetch(DATA_URL);
const data = await response.json();
allChampions = Object.values(data.data);
displayChampions(allChampions);
document.getElementById("loading").style.display = "none";
}
catch (error) {
document.getElementById("loading").innerHTML =
"❌ Impossible de charger les champions.<br><br>" +
"Vérifie que tu as Internet puis recharge la page.";
console.error(error);
}
}
/*
=========================================
AFFICHER LES CHAMPIONS
=========================================
*/
function displayChampions(champions) {
const container =
document.getElementById("champions");
container.innerHTML = "";
if (champions.length === 0) {
container.innerHTML =
`<div style="
text-align:center;
grid-column:1/-1;
padding:50px;
color:#c8aa6e;
">
Aucun champion trouvé.
</div>`;
return;
}
champions.forEach(champion => {
const card =
document.createElement("div");
card.className = "card";
const tags =
champion.tags
.map(tag => `<span class="tag">${translateRole(tag)}</span>`)
.join("");
card.innerHTML = `
<img
src="${SPLASH_URL}${champion.id}_0.jpg"
alt="${champion.name}"
loading="lazy"
>
<div class="card-content">
<h3>
${champion.name}
</h3>
<p class="title">
${champion.title}
</p>
<div class="tags">
${tags}
</div>
</div>
`;
card.addEventListener("click", () => {
openChampion(champion);
});
container.appendChild(card);
});
}
/*
=========================================
TRADUIRE LES ROLES
=========================================
*/
function translateRole(role) {
const roles = {
Assassin: "󰢊 Assassin",
Mage: "🔮 Mage",
Tank: " Tank",
Fighter: "󱟋 Combattant",
Marksman: "🏹 Tireur",
Support: "✨ Support"
};
return roles[role] || role;
}
/*
=========================================
FILTRE PAR ROLE
=========================================
*/
function filterRole(role, button) {
currentRole = role;
document
.querySelectorAll(".filters button")
.forEach(btn => {
btn.classList.remove("active");
});
button.classList.add("active");
applyFilters();
}
/*
=========================================
RECHERCHE
=========================================
*/
document
.getElementById("search")
.addEventListener("input", applyFilters);
function applyFilters() {
const search =
document
.getElementById("search")
.value
.toLowerCase();
let result =
allChampions.filter(champion => {
const matchesSearch =
champion.name
.toLowerCase()
.includes(search);
let matchesRole = true;
if (currentRole !== "Tous") {
matchesRole =
champion.tags.includes(
currentRole === "Combattant"
? "Fighter"
: currentRole === "Tireur"
? "Marksman"
: currentRole
);
}
return matchesSearch && matchesRole;
});
displayChampions(result);
}
/*
=========================================
OUVRIR CHAMPION
=========================================
*/
async function openChampion(champion) {
const modal =
document.getElementById("modal");
modal.classList.add("show");
document.getElementById("modalName").textContent =
champion.name;
document.getElementById("modalTitle").textContent =
champion.title;
document.getElementById("modalSplash").src =
`${SPLASH_URL}${champion.id}_0.jpg`;
document.getElementById("modalDescription").textContent =
champion.blurb;
const stats =
champion.stats;
const statsHTML = `
<div class="stat">
<div class="stat-name">󰥩 PV</div>
<div class="stat-value">${stats.hp}</div>
</div>
<div class="stat">
<div class="stat-name">💙 Mana</div>
<div class="stat-value">${stats.mp}</div>
</div>
<div class="stat">
<div class="stat-name">󱟋 Attaque</div>
<div class="stat-value">${stats.attackdamage}</div>
</div>
<div class="stat">
<div class="stat-name"> Armure</div>
<div class="stat-value">${stats.armor}</div>
</div>
<div class="stat">
<div class="stat-name">🔮 Résistance magique</div>
<div class="stat-value">${stats.spellblock}</div>
</div>
<div class="stat">
<div class="stat-name">🏃 Vitesse</div>
<div class="stat-value">${stats.movespeed}</div>
</div>
<div class="stat">
<div class="stat-name">🎯 Portée</div>
<div class="stat-value">${stats.attackrange}</div>
</div>
<div class="stat">
<div class="stat-name">⚡ Vitesse attaque</div>
<div class="stat-value">${stats.attackspeed}</div>
</div>
`;
document.getElementById("modalStats").innerHTML =
statsHTML;
/*
=====================================
CHARGER LES COMPETENCES
=====================================
*/
const detailURL =
`https://ddragon.leagueoflegends.com/cdn/${VERSION}/data/fr_FR/champion/${champion.id}.json`;
try {
const response =
await fetch(detailURL);
const data =
await response.json();
const detailed =
data.data[champion.id];
let skillsHTML = "";
/*
PASSIF
*/
skillsHTML += `
<div class="skill">
<img
src="${IMAGE_URL}${detailed.passive.image.full}"
>
<h4>
Passif :
${detailed.passive.name}
</h4>
<p>
${detailed.passive.description}
</p>
</div>
`;
/*
SORTS
*/
detailed.spells.forEach((spell, index) => {
skillsHTML += `
<div class="skill">
<img
src="${IMAGE_URL}${spell.image.full}"
>
<h4>
${spell.name}
</h4>
<p>
${spell.description}
</p>
</div>
`;
});
document.getElementById("modalSkills").innerHTML =
skillsHTML;
}
catch (error) {
document.getElementById("modalSkills").innerHTML =
"<p>Impossible de charger les compétences.</p>";
}
}
/*
=========================================
FERMER MODALE
=========================================
*/
function closeModal() {
document
.getElementById("modal")
.classList.remove("show");
}
/*
=========================================
CLIQUER EN DEHORS
=========================================
*/
document
.getElementById("modal")
.addEventListener("click", function(event) {
if (event.target === this) {
closeModal();
}
});
/*
=========================================
ECHAP
=========================================
*/
document.addEventListener("keydown", function(event) {
if (event.key === "Escape") {
closeModal();
}
});
/*
=========================================
LANCER LE SITE
=========================================
*/
loadChampions();
